cmake_minimum_required(VERSION 3.22)
project(decision_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/llama.cpp)
add_library(decision STATIC src/decision.cpp src/media.cpp)
target_include_directories(decision PUBLIC include PRIVATE third_party/llama.cpp/src third_party/llama.cpp/vendor)
target_link_libraries(decision PUBLIC llama mtmd)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(d1-cli src/cli.cpp)
  target_include_directories(d1-cli PRIVATE third_party/llama.cpp/vendor)
  target_link_libraries(d1-cli PRIVATE decision)
endif()
